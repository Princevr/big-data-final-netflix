-- Movies vs TV Shows
SELECT type, COUNT(*) AS num_titles
FROM netflix_clean
GROUP BY type;

-- Top 10 countries
SELECT country, COUNT(*) AS num_titles
FROM netflix_clean
WHERE country IS NOT NULL
GROUP BY country
ORDER BY num_titles DESC
LIMIT 10;

-- Titles per year
SELECT YEAR(data_added) AS year_added, COUNT(*) AS num_titles
FROM netflix_clean
WHERE data_added IS NOT NULL
GROUP BY YEAR(data_added)
ORDER BY year_added;
